    detectEnemy(enemyNodeId) { this.enemyNode = this.map[enemyNodeId]; }

   

    decideTactic() {

      if (!this.enemyNode) return "DIRECT_PUSH";

      const node = this.enemyNode;

      if (node.data.elevation > 1) return "SUPPRESS_FLANK";

      if (node.data.cover > 0.7) return "SPLIT_PUSH";

      if (node.data.risk > 4) return "BAIT_AND_SWITCH";

      return "DIRECT_PUSH";

    }

    executeTactic() {

      const tactic = this.decideTactic();

      const node = this.enemyNode;

      if (!node) return;

     

      switch (tactic) {

        case "SUPPRESS_FLANK":

          this.players.forEach(p => {

            if (p.role === "ANCHOR") p.target = this.getHighGround() || node;

            else if (p.role === "ENTRY") p.target = node;

            else p.target = this.getSideNode(node);

          });

          break;

        case "SPLIT_PUSH":

          this.players.forEach(p => {

            if (p.role === "IGL" || p.role === "ENTRY") p.target = node;

            else p.target = this.getSideNode(node);

          });

          break;

        case "BAIT_AND_SWITCH":

          this.players.forEach(p => {

            if (p.role === "SUPPORT") p.target = node;

            else p.target = this.getSideNode(node);

          });

          break;

        default:

          this.players.forEach(p => { p.target = node; });

      }

    }

    getSideNode(node) {

      const neighbors = node.neighbors;

      return this.map[neighbors[0]];

    }

  function monitorSpeaking(stream, unitId) {

    const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

    const source = audioCtx.createMediaStreamSource(stream);

    const analyser = audioCtx.createAnalyser();

    analyser.fftSize = 256;

    function checkVolume() {

      if (!isVoiceConnected) return; // 如果斷線就停止監聽

      analyser.getByteFrequencyData(dataArray);

      let sum = 0;

      let avg = sum / dataArray.length;

        // 音量大於一定門檻，加上 'speaking' class 觸發閃爍

          unitEl.classList.add('speaking');

        } else {

          unitEl.classList.remove('speaking');

        }

      }

      requestAnimationFrame(checkVolume);

    }

  function bindMapControls() {

    container.addEventListener('wheel', e => { e.preventDefault(); applyZoom(e.deltaY > 0 ? -0.1 : 0.1); }, { passive: false });

    let isMoving = false, startX, startY; let initialPinchDist = null, initialZoom = 1; let lastTap = 0;

    const onStart = e => {

      if (isLayoutLocked || isRoutingMode || e.target.classList.contains('unit-drag')) return;

      if (e.touches && e.touches.length === 2) { initialPinchDist = Math.hypot(e.touches[0].clientX - e.touches[1].clientX, e.touches[0].clientY - e.touches[1].clientY); initialZoom = zoom; isMoving = false; return; }

      isMoving = true; startX = (e.touches ? e.touches[0].clientX : e.clientX) - panX; startY = (e.touches ? e.touches[0].clientY : e.clientY) - panY;

    };

    const onMove = e => {

      if (isLayoutLocked || (!isMoving && initialPinchDist === null)) return;

      if(e.cancelable) e.preventDefault();

      if (e.touches && e.touches.length === 2 && initialPinchDist) { const dist = Math.hypot(e.touches[0].clientX - e.touches[1].clientX, e.touches[0].clientY - e.touches[1].clientY); zoom = Math.min(Math.max(1, initialZoom * (dist / initialPinchDist)), 4); updateTransform(); return; }

      if (isMoving) { panX = (e.touches ? e.touches[0].clientX : e.clientX) - startX; panY = (e.touches ? e.touches[0].clientY : e.clientY) - startY; updateTransform(); }

    };

    const onEnd = e => {

      if (e.touches && e.touches.length < 2) initialPinchDist = null; if (!e.touches || e.touches.length === 0) isMoving = false;

      const currentTime = new Date().getTime(); const tapLength = currentTime - lastTap;

      if (!isLayoutLocked && tapLength < 300 && tapLength > 0 && e.target.id !== 'sz-slider' && e.target.id !== 'unit-size-slider') { zoom = 1; panX = 0; panY = 0; updateTransform(); if(e.cancelable) e.preventDefault(); }

      lastTap = currentTime;

    };

    container.addEventListener('dblclick', (e) => { if(isLayoutLocked || e.target.classList.contains('unit-drag')) return; zoom = 1; panX = 0; panY = 0; updateTransform(); });

    container.addEventListener('mousedown', onStart); container.addEventListener('touchstart', onStart, { passive: false });

    document.addEventListener('mousemove', onMove, { passive: false }); document.addEventListener('touchmove', onMove, { passive: false });

    document.addEventListener('mouseup', onEnd); document.addEventListener('touchend', onEnd);

  }

  function updateTransform() {

    wrapper.style.transform = `translate(${panX}px, ${panY}px) scale(${zoom})`; document.getElementById('z-val').innerText = Math.round(zoom * 100) + '%';

  }

        if(isolated.length > 0) { score -= 20; tips.push(`<div class="ai-item" style="border-left:3px solid var(--enemy);">💀 <b>[-20] 孤狼送頭：</b>${isolated.join(', ')} 脫離隊友補槍距離 (超過 45m)！倒地保證無法救援。</div>`); }

        else { tips.push(`<div class="ai-item" style="border-left:3px solid #00ffcc;">🛡️ <b>聯動補槍：</b>陣型緊密聯動，可瞬間形成人數優勢。</div>`); }

        if (enemyCenters.length > 1) {

          let closestTeam = null;

          let minDistToTeam = Infinity;

          let teamsWithin100m = 0;

          enemyCenters.forEach(team => {

            let dist = Math.hypot(fC.x - team.c.x, fC.y - team.c.y) * 10;

            if (dist < minDistToTeam) { minDistToTeam = dist; closestTeam = team; }

            if (dist < 100) teamsWithin100m++;

          });

          if (closestTeam) {

            tips.push(`<div class="ai-item" style="border-left:3px solid ${closestTeam.color}; background:rgba(255,255,255,0.05);">🚨 <b>首要威脅鎖定：</b>距離最近的為 <b style="color:${closestTeam.color}">${closestTeam.name}</b> (${Math.round(minDistToTeam)}m)。請指揮官優先防範其突擊！</div>`);

          }

          if (teamsWithin100m >= 2) {

            score -= 20;

            tips.push(`<div class="ai-item" style="border-left:3px solid var(--enemy);">⚠️ <b>[-20] 絞肉機中心 (防守死局)：</b>您正被 ${teamsWithin100m} 支隊伍近距離包夾！建議：立刻收縮陣型，尋找防守掩體死角，不要主動開火暴露位置，讓敵方互相消耗 (勸架)。</div>`);

          } else if (teamsWithin100m === 1 && enemyCenters.length >= 2) {

            score += 15;

            tips.push(`<div class="ai-item" style="border-left:3px solid #00ffcc;">🦅 <b>[+15] 完美外圍黃雀 (進攻佈局)：</b>您位於多方交火區外圍！建議：保持隱蔽，觀察其他隊伍互咬，等擊倒信息出現後，採取 2-2 分線從側翼切入收割！</div>`);

          }

        } else {

          const rsh1 = f.find(u => u.role === 'Rusher' || u.role === 'IGL'); const rsh2 = [...f].reverse().find(u => u.role === 'Rusher' || u.role === 'Support');

          if (rsh1 && rsh2 && rsh1.id !== rsh2.id) {

            const dx1 = rsh1.x - eC.x, dy1 = rsh1.y - eC.y; const dx2 = rsh2.x - eC.x, dy2 = rsh2.y - eC.y;

            const dotProduct = (dx1 * dx2) + (dy1 * dy2); const mag1 = Math.hypot(dx1, dy1), mag2 = Math.hypot(dx2, dy2);

            const angle = Math.acos(dotProduct / (mag1 * mag2)) * (180 / Math.PI);

            if (angle >= 60 && angle <= 135) { tips.push(`<div class="ai-item" style="border-left:3px solid #00ffcc;">⚔️ <b>死亡交叉火力：</b>雙槍線形成 ${Math.round(angle)}° 完美夾角！</div>`); }

            else if (angle < 40) { score -= 15; tips.push(`<div class="ai-item" style="border-left:3px solid #ffcc00;">⚠️ <b>[-15] 火力線重疊：</b>進攻夾角僅 ${Math.round(angle)}°，對手只需躲在掩體後就能防死你們。</div>`); }

            else { score -= 5; tips.push(`<div class="ai-item" style="border-left:3px solid #ffcc00;">⚠️ <b>[-5] 夾角過大：</b>進攻角度 ${Math.round(angle)}° 偏大，容易被各個擊破。</div>`); }

          }

        }

        const sniper = f.find(u => u.role === 'Sniper');

        let avgDistToEnemy_m = (f.reduce((sum, u) => sum + Math.hypot(u.x - eC.x, u.y - eC.y), 0) / 4) * 10;

        let isCQC = avgDistToEnemy_m < 40;

        if (sniper) {

          const sniperDist_m = Math.hypot(sniper.x - eC.x, sniper.y - eC.y) * 10;

          if (sniperDist_m > 60) { tips.push(`<div class="ai-item" style="border-left:3px solid #00ffcc;">🎯 <b>狙擊高點：</b>狙擊手距離達 ${Math.round(sniperDist_m)}m，可提供絕對火力壓制與視野。</div>`); }

          else if (sniperDist_m < 40 && isCQC) { tips.push(`<div class="ai-item" style="border-left:3px solid #ffaa00;">🎯 <b>衝鋒破點狙：</b>極近距離(${Math.round(sniperDist_m)}m)狙擊！高風險高報酬的特種打法。</div>`); }

          else { score -= 10; tips.push(`<div class="ai-item" style="border-left:3px solid var(--enemy);">❌ <b>[-10] 狙擊手曝險：</b>站位不三不四(${Math.round(sniperDist_m)}m)，極易被步槍點殘。</div>`); }

        }

        currentAIScore = Math.min(100, Math.max(0, score));

        let scoreColor = currentAIScore >= 90 ? '#00ffcc' : (currentAIScore >= 60 ? '#ffcc00' : 'var(--enemy)');

        scoreDisplay.style.color = scoreColor; scoreDisplay.innerText = currentAIScore;

       

        c.innerHTML = tips.length ? tips.join("") : `<div class="ai-item">✅ 陣型配置無懈可擊，請保持。</div>`;

      } catch (err) {

        c.innerHTML = `<div class="ai-item" style="border-left:3px solid #ff4444; color:#ff4444;">❌ <b>系統發生預期外錯誤，請通知管理員：</b><br>${err.message}</div>`;

        scoreDisplay.style.color = '#ff4444'; scoreDisplay.innerText = "ERR";

      }

    }, 1200);

  }
